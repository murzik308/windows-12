function copy(text,btn){
    navigator.clipboard.writeText(text);

    const old = btn.innerText;
    btn.innerText = "Copied!";
    btn.style.background = "#22c55e";

    setTimeout(()=>{
        btn.innerText = old;
        btn.style.background = "";
    },1200);
